   147→   147→  _setupExpress() {
   148→   148→    this.app = express();
   149→   149→
   150→   150→    // Trust proxy — required because we sit behind Nginx/Caddy.
   151→   151→    // Without this, express-rate-limit sees the proxy IP (172.x.x.x)

   154→   154→    this.app.set('trust proxy', 1);
   155→   155→
   156→   156→    // Security middleware
   157→   157→    this.app.use(helmet());

   159→   159→      origin: config.corsOrigins === '*' ? true : config.corsOrigins.split(','),
   160→   160→    }));
   161→   161→
   162→   162→    // Body parsing — preserve raw body for webhook signature verification (PawaPay Content-Digest)
   163→   163→    this.app.use(express.json({
   164→   164→      limit: '1mb',
   165→   165→      verify: (req, _res, buf) => { req.rawBody = buf.toString(); },
   166→   166→    }));
   167→   167→    this.app.use(express.urlencoded({ extended: true }));
   168→   168→
   169→   169→    // Compression

   301→   301→      // Passer wasOfflineNotified au positionHandler pour qu'il déclenche
   302→   302→      // la notif "back online" si besoin (sans dépendre du callback qui lirait

   304→   304→      await this.positionHandler.handle(position, socketId, { wasOfflineNotified });
   305→   305→    } catch (err) {
   306→   306→      logger.error(`Error handling position from ${socketId}:`, err);
   307→   307→    }

   317→   317→  /**
   318→   318→   * Handle device TCP disconnection.
   319→   319→   *
   320→   320→   * Strategy: mark is_connected = false in DB ONLY (Flutter sees "offline" instantly
   321→   321→   * via Supabase Realtime). No notification sent here — 90% of TCP disconnects are
   322→   322→   * micro-coupures (antenna switch, tunnel) and would cause spam.
   323→   323→   *
   324→   324→   * Real disconnection notification is delegated to _cleanupStaleConnections()
   325→   325→   * which confirms offline status after 5 min without data and sends via sendAlert()
   326→   326→   * (push + SMS + WhatsApp, multi-user, with credit gate).
   327→   327→   *
   328→   328→   * IMPORTANT: This method does NOT handle acc_on/acc_off (alimentation) — those are
   329→   329→   * evaluated in alertEngine.js from GPS position data (accStatus field), completely
   330→   330→   * independent from TCP socket state.
   331→   331→   */
   332→   332→  async _onDeviceDisconnect(socketId, imei) {
   333→   333→    if (!imei) return;
   334→   334→    logger.info(`[DISCONNECT] Device ${imei} disconnected (socket ${socketId})`);
   335→   335→    try {
   336→   336→      const supabase = getSupabaseService();
   337→   337→      await supabase.markDeviceDisconnected(imei);
   338→   338→
   339→   339→      // ═══════════════════════════════════════════════════════════════
   340→   340→      // ADAPTATIVE DEBOUNCE — Timer selon le contexte véhicule
   341→   341→      // ═══════════════════════════════════════════════════════════════
   342→   342→      // Logique (proposition user + anti-fausse-alerte parking) :
   343→   343→      //
   344→   344→      // 1. Véhicule EN MARCHE (acc_status=true) → 5 min
   345→   345→      //    Probabilité : tunnel, pont, zone blanche → on attend pour éviter spam
   346→   346→      //
   347→   347→      // 2. Véhicule ARRÊTÉ DEPUIS > 5 min (acc_status=false, last_update vieux)
   348→   348→      //    Probabilité : sabotage, batterie morte → on alerte vite (30s)
   349→   349→      //
   350→   350→      // 3. Véhicule ARRÊTÉ RÉCEMMENT (< 5 min après acc_status→false)
   351→   351→      //    Probabilité : stationnement normal, l'utilisateur vient de garer
   352→   352→      //    → on attend 5 min pour éviter fausse alerte au parking
   353→   353→      // ═══════════════════════════════════════════════════════════════
   354→   354→
   355→   355→      // Annuler un timer existant pour cet IMEI
   356→   356→      if (this._disconnectTimers?.has(imei)) {
   357→   357→        clearTimeout(this._disconnectTimers.get(imei));
   358→   358→      }
   359→   359→      if (!this._disconnectTimers) this._disconnectTimers = new Map();
   360→   360→
   361→   361→      // Récupérer l'état ACC et last_update pour déterminer le délai adaptatif
   362→   362→      const { data: currentPos } = await supabase.client
   363→   363→        .from('last_known_positions')
   364→   364→        .select('acc_status, last_update')
   365→   365→        .eq('device_id', imei)
   366→   366→        .maybeSingle();
   367→   367→
   368→   368→      const accStatus = currentPos?.acc_status;
   369→   369→      const lastUpdateMs = currentPos?.last_update
   370→   370→        ? Date.now() - new Date(currentPos.last_update).getTime()
   371→   371→        : null;
   372→   372→
   373→   373→      // Déterminer le délai selon l'état du véhicule (logique adaptative)
   374→   374→      let debounceDelay;
   375→   375→      let delayReason;
   376→   376→
   377→   377→      if (accStatus === true) {
   378→   378→        // Véhicule en marche → tunnel/pont probable → 5 min
   379→   379→        debounceDelay = 5 * 60 * 1000;
   380→   380→        delayReason = 'vehicle_running_5min';
   381→   381→      } else if (accStatus === false && lastUpdateMs !== null && lastUpdateMs > 5 * 60 * 1000) {
   382→   382→        // Véhicule arrêté depuis > 5 min → sabotage probable → 30s
   383→   383→        debounceDelay = 30 * 1000;
   384→   384→        delayReason = 'vehicle_parked_long_30s';
   385→   385→      } else {
   386→   386→        // Véhicule arrêté récemment (< 5 min) ou état inconnu → stationnement normal → 5 min
   387→   387→        debounceDelay = 5 * 60 * 1000;
   388→   388→        delayReason = 'vehicle_parked_recent_5min';
   389→   389→      }
   390→   390→
   391→   391→      logger.info(`[DISCONNECT] ${imei}: marked offline, ${delayReason} timer started (${debounceDelay / 1000}s) — acc=${accStatus}, lastUpdateAge=${lastUpdateMs ? Math.round(lastUpdateMs / 1000) + 's' : 'null'}`);
   392→   392→
   393→   393→      // Démarrer le timer adaptatif
   394→   394→      const timer = setTimeout(async () => {
   395→   395→        try {
   396→   396→          // Double-check device is still offline (didn't reconnect during 30s)
   397→   397→          const { data: pos } = await supabase.client
   398→   398→            .from('last_known_positions')
   399→   399→            .select('is_connected, last_update')
   400→   400→            .eq('device_id', imei)
   401→   401→            .maybeSingle();
   402→   402→
   403→   403→          if (pos?.is_connected === true) {
   404→   404→            // Device reconnected during the 30s window — skip notification
   405→   405→            logger.info(`[DISCONNECT] ${imei}: reconnected within 30s debounce, skipping offline notification`);
   406→   406→            this._disconnectTimers.delete(imei);
   407→   407→            return;
   408→   408→          }
   409→   409→
   410→   410→          // Check if we already notified (anti-spam)
   411→   411→          if (this._disconnectNotified.has(imei)) {
   412→   412→            logger.debug(`[DISCONNECT] ${imei}: already notified, skipping`);
   413→   413→            this._disconnectTimers.delete(imei);
   414→   414→            return;
   415→   415→          }
   416→   416→
   417→   417→          // Send HIGH PRIORITY notification
   418→   418→          const deviceName = await supabase.getDeviceName(imei) || imei;
   419→   419→          const allUsers = await supabase.getDeviceAllUsers(imei);
   420→   420→
   421→   421→          if (!allUsers || allUsers.length === 0) {
   422→   422→            logger.info(`[DISCONNECT] ${imei}: no linked users, skipping notification`);
   423→   423→            this._disconnectTimers.delete(imei);
   424→   424→            return;
   425→   425→          }
   426→   426→
   427→   427→          const notificationService = getNotificationService();
   428→   428→          const result = await notificationService.sendAlert({
   429→   429→            users: allUsers.map(u => ({ userId: u.userId })),
   430→   430→            type: 'device_offline',
   431→   431→            title: '🚨 Capteur déconnecté',
   432→   432→            body: `${deviceName} est hors ligne — Vérifiez votre véhicule`,
   433→   433→            deviceId: imei,
   434→   434→            data: {
   435→   435→              type: 'device_offline',
   436→   436→              device_id: imei,
   437→   437→              source: 'tcp_disconnect_debounced',
   438→   438→              priority: 'critical',
   439→   439→            },
   440→   440→            priority: 'critical', // FIX #3: signal to notificationService to use MAX priority
   441→   441→          });
   442→   442→
   443→   443→          // Mark as notified
   444→   444→          this._disconnectNotified.add(imei);
   445→   445→
   446→   446→          const channelSummary = result
   447→   447→            .map(r => {
   448→   448→              const chans = (r.results || []).map(cr => `${cr.channel}=${cr.status}`).join(',');
   449→   449→              return `user=${r.userId.substring(0, 8)} [${chans}]`;
   450→   450→            })
   451→   451→            .join(' | ');
   452→   452→          logger.info(`[DISCONNECT] ${imei}: sent offline notification (${delayReason}, ${debounceDelay / 1000}s debounce) → ${channelSummary}`);
   453→   453→        } catch (err) {
   454→   454→          logger.error(`[DISCONNECT] ${imei}: debounced notification failed:`, err.message);
   455→   455→        } finally {
   456→   456→          this._disconnectTimers.delete(imei);
   457→   457→        }
   458→   458→      }, debounceDelay);
   459→   459→
   460→   460→      this._disconnectTimers.set(imei, timer);
   461→   461→
   462→   462→      // Clear back-online check cache so next reconnect triggers fresh check
   463→   463→      if (this.positionHandler) {
   464→   464→        this.positionHandler.clearBackOnlineCheck(imei);
   465→   465→      }
   466→   466→    } catch (err) {
   467→   467→      logger.error(`[DISCONNECT] Failed for ${imei}:`, err);
   468→   468→    }
   469→   469→  }
   470→   470→
   471→   471→  /**
   472→   472→   * Cancel the disconnect debounce timer when a position arrives (device reconnected).
   473→   473→   * Called from _onPosition() in server.js.
   474→   474→   */
   475→   475→  _cancelDisconnectDebounce(imei) {
   476→   476→    if (this._disconnectTimers?.has(imei)) {
   477→   477→      clearTimeout(this._disconnectTimers.get(imei));
   478→   478→      this._disconnectTimers.delete(imei);
   479→   479→      logger.info(`[RECONNECT] ${imei}: cancelled pending disconnect notification (micro-coupure)`);
   480→   480→      return true;
   481→   481→    }
   482→   482→    return false;
   483→   483→  }
   484→   484→
   485→   485→  /**
   486→   486→   * Setup periodic cleanup tasks
   487→   487→   * - Alert engine dedup cleanup (existing — unchanged)
   488→   488→   * - Alert engine device state cleanup (existing — unchanged)
   489→   489→   * - Stale connection cleanup: detects devices offline > 5 min, sends
   490→   490→   *   notification via sendAlert() (push + SMS + WhatsApp) to all linked users
   491→   491→   */
   492→   492→  _setupCleanup() {
   493→   493→    // Run cleanup every 2 minutes
   494→   494→    this._cleanupInterval = setInterval(() => {
   495→   495→      try {
   496→   496→        const alertEngine = getAlertEngine();
   497→   497→        alertEngine.cleanupDedup();
   498→   498→        alertEngine.cleanupDeviceStates();
   499→   499→
   500→   500→        // Detect devices with no data for 5+ min, notify via all channels
   501→   501→        this._cleanupStaleConnections();
   502→   502→
   503→   503→        logger.info('Periodic cleanup completed');
   504→   504→      } catch (err) {
   505→   505→        logger.error('Cleanup failed:', err);
   506→   506→      }
   507→   507→    }, 2 * 60 * 1000); // Every 2 minutes
   508→   508→  }
   509→   509→
   510→   510→  /**
   511→   511→   * Detect devices truly offline (no data for 5+ min) and send notification
   512→   512→   * via the unified sendAlert() — covers push + SMS + WhatsApp for all linked users.
   513→   513→   *
   514→   514→   * Covers two scenarios:
   515→   515→   * 1. TCP close fired but notification was deferred (see _onDeviceDisconnect)
   516→   516→   * 2. TCP close never fired (silent network drop — e.g. device powered off)
   517→   517→   *
   518→   518→   * Anti-spam: _disconnectNotified Set prevents re-notifying the same IMEI
   519→   519→   * while it remains offline. Cleared instantly on reconnection in _onPosition().
   520→   520→   *
   521→   521→   * NOTE: This only handles device_offline (network disconnect). It does NOT
   522→   522→   * handle acc_on/acc_off (alimentation) — those are in alertEngine.js.
   523→   523→   */
   524→   524→  async _cleanupStaleConnections() {
   525→   525→    try {
   526→   526→      const supabase = getSupabaseService();
   527→   527→      const notificationService = getNotificationService();
   528→   528→
   529→   529→      // Find devices still marked connected but with no data for 5 minutes
   530→   530→      const cutoff = new Date(Date.now() - 5 * 60 * 1000).toISOString();
   531→   531→      const { data: staleDevices, error } = await supabase.client
   532→   532→        .from('last_known_positions')
   533→   533→        .select('device_id')
   534→   534→        .eq('is_connected', true)
   535→   535→        .lt('last_update', cutoff);
   536→   536→
   537→   537→      if (error || !staleDevices || staleDevices.length === 0) {
   538→   538→        return; // No stale devices — silent exit (no log spam)
   539→   539→      }
   540→   540→
   541→   541→      // Mark them all disconnected in DB
   542→   542→      const staleImeis = staleDevices.map(d => d.device_id);
   543→   543→      const { error: updateErr } = await supabase.client
   544→   544→        .from('last_known_positions')
   545→   545→        .update({ is_connected: false })
   546→   546→        .in('device_id', staleImeis);
   547→   547→
   548→   548→      if (updateErr) {
   549→   549→        logger.error('Failed to mark stale devices disconnected:', updateErr);
   550→   550→        return;
   551→   551→      }
   552→   552→
   553→   553→      logger.info(`🧹 Stale cleanup: ${staleDevices.length} device(s) offline > 5min: ${staleImeis.join(', ')}`);
   554→   554→
   555→   555→      // Send notification via unified sendAlert() for each non-duplicate device
   556→   556→      for (const { device_id: imei } of staleDevices) {
   557→   557→        // Anti-spam: skip if already notified for this offline period
   558→   558→        if (this._disconnectNotified.has(imei)) {
   559→   559→          logger.debug(`[STALE] ${imei}: device_offline notification SKIPPED (already notified, awaiting reconnection)`);
   560→   560→          continue;
   561→   561→        }
   562→   562→
   563→   563→        try {
   564→   564→          // Get device name for human-readable notification body
   565→   565→          const deviceName = await supabase.getDeviceName(imei) || imei;
   566→   566→
   567→   567→          // Get ALL users linked to this device (owner + shared users)
   568→   568→          const allUsers = await supabase.getDeviceAllUsers(imei);
   569→   569→          if (!allUsers || allUsers.length === 0) {
   570→   570→            logger.info(`[STALE] ${imei}: no linked users found, skipping notification`);
   571→   571→            continue;
   572→   572→          }
   573→   573→
   574→   574→          // Send via unified sendAlert() — handles channel selection, credit gate,
   575→   575→          // DB insert, push + SMS + WhatsApp per user preference
   576→   576→          const result = await notificationService.sendAlert({
   577→   577→            users: allUsers.map(u => ({ userId: u.userId })),
   578→   578→            type: 'device_offline',
   579→   579→            title: 'Capteur déconnecté',
   580→   580→            body: `${deviceName} est hors ligne`,
   581→   581→            deviceId: imei,
   582→   582→            data: { type: 'device_offline', device_id: imei, source: 'stale_cleanup' },
   583→   583→          });
   584→   584→
   585→   585→          // Mark as notified to prevent re-sending while device stays offline
   586→   586→          this._disconnectNotified.add(imei);
   587→   587→
   588→   588→          const channelSummary = result
   589→   589→            .map(r => {
   590→   590→              const chans = (r.results || []).map(cr => `${cr.channel}=${cr.status}`).join(',');
   591→   591→              return `user=${r.userId.substring(0, 8)} [${chans}]`;
   592→   592→            })
   593→   593→            .join(' | ');
   594→   594→          logger.info(`[STALE] ${imei}: sendAlert result → ${channelSummary}`);
   595→   595→        } catch (err) {
   596→   596→          logger.error(`[STALE] Notification failed for ${imei}:`, err.message);
   597→   597→          // Don't add to _disconnectNotified on error — retry on next cycle
   598→   598→        }
   599→   599→      }
   600→   600→    } catch (err) {
   601→   601→      logger.error('Stale connection cleanup failed:', err);
   602→   602→    }
   603→   603→  }
   604→   604→
   605→   605→  /**
   606→   606→   * Setup graceful shutdown handlers
   607→   607→   */
   608→   608→  _setupShutdown() {
   609→   609→    const shutdown = async (signal) => {
   610→   610→      if (this._shuttingDown) return;
   611→   611→      this._shuttingDown = true;
   612→   612→
   613→   613→      logger.info(`Received ${signal}, shutting down gracefully...`);
   614→   614→
   615→   615→      // Clear cleanup interval
   616→   616→      if (this._cleanupInterval) {
   617→   617→        clearInterval(this._cleanupInterval);
   618→   618→      }
   619→   619→
   620→   620→      // Stop Realtime listener
   621→   621→      const notificationService = getNotificationService();
   622→   622→      notificationService.stopRealtimeListener();
   623→   623→
   624→   624→      // Stop Parking Alert Loops (bip-bip critique)
   625→   625→      // Important : arrête les boucles proprement au shutdown pour éviter
   626→   626→      // les résidus au redémarrage.
   627→   627→      try {
   628→   628→        const { getParkingAlertLoop } = require('./services/parkingAlertLoop');
   629→   629→        getParkingAlertLoop().stopAll();
   630→   630→      } catch (e) {
   631→   631→        logger.warn('ParkingAlertLoop stop failed (non-critical):', e.message);
   632→   632→      }
   633→   633→
   634→   634→      // Stop TCP server
   635→   635→      if (this.tcpServer) {
   636→   636→        await this.tcpServer.stop();
   637→   637→      }
   638→   638→
   639→   639→      // Stop HTTP server
   640→   640→      if (this.httpServer) {
   641→   641→        this.httpServer.close(() => {
   642→   642→          logger.info('HTTP server closed');
   643→   643→          process.exit(0);
   644→   644→        });
   645→   645→
   646→   646→        // Force exit after 10 seconds
   647→   647→        setTimeout(() => {
   648→   648→          logger.error('Forced shutdown after timeout');
   649→   649→          process.exit(1);
   650→   650→        }, 10000);
   651→   651→      } else {
   652→   652→        process.exit(0);
   653→   653→      }
   654→   654→    };
   655→   655→
   656→   656→    process.on('SIGTERM', () => shutdown('SIGTERM'));
   657→   657→    process.on('SIGINT', () => shutdown('SIGINT'));
   658→   658→    process.on('SIGQUIT', () => shutdown('SIGQUIT'));
   659→   659→
   660→   660→    // Handle uncaught errors
   661→   661→    process.on('uncaughtException', (err) => {
   662→   662→      logger.error('Uncaught exception:', err);
   663→   663→      shutdown('uncaughtException');
   664→   664→    });
   665→   665→
   666→   666→    process.on('unhandledRejection', (reason) => {
   667→   667→      logger.error('Unhandled rejection:', reason);
   668→   668→      shutdown('unhandledRejection');
   669→   669→    });
   670→   670→  }
   671→   671→}
