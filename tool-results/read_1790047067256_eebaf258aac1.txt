   144→  /**
   145→   * Setup Express HTTP server
   146→   */
   147→  _setupExpress() {
   148→    this.app = express();
   149→
   150→    // Trust proxy — required because we sit behind Nginx/Caddy.
   151→    // Without this, express-rate-limit sees the proxy IP (172.x.x.x)
   152→    // for ALL clients and the X-Forwarded-For warning fires.
   153→    // '1' = trust the first proxy hop (typical for a single reverse proxy).
   154→    this.app.set('trust proxy', 1);
   155→
   156→    // Security middleware
   157→    this.app.use(helmet());
   158→    this.app.use(cors({
   159→      origin: config.corsOrigins === '*' ? true : config.corsOrigins.split(','),
   160→    }));
   161→
   162→    // Body parsing — preserve raw body for webhook signature verification (PawaPay Content-Digest)
   163→    this.app.use(express.json({
   164→      limit: '1mb',
   165→      verify: (req, _res, buf) => { req.rawBody = buf.toString(); },
   166→    }));
   167→    this.app.use(express.urlencoded({ extended: true }));
   168→
   169→    // Compression
   170→    this.app.use(compression());
   171→
   172→    // Rate limiting (30 requests per minute per IP)
   173→    const limiter = rateLimit({
   174→      windowMs: 60 * 1000, // 1 minute
   175→      max: 30,
   176→      standardHeaders: true,
   177→      legacyHeaders: false,
   178→      message: { error: 'Too many requests, please try again later' },
   179→      skip: (req) => req.path === '/api/health', // Don't rate-limit health checks
   180→    });
   181→    this.app.use('/api', limiter);
   182→
   183→    // Stricter rate limit for webhooks (10/min — prevents webhook flooding)
   184→    const webhookLimiter = rateLimit({
   185→      windowMs: 60 * 1000,
   186→      max: 10,
   187→      standardHeaders: true,
   188→      legacyHeaders: false,
   189→      message: { error: 'Webhook rate limit exceeded' },
   190→    });
   191→    this.app.use('/webhooks', webhookLimiter);
   192→
   193→    // Request logging
   194→    if (config.isDev) {
   195→      this.app.use(morgan('dev'));
   196→    } else {
   197→      this.app.use(morgan('combined', {
   198→        stream: { write: (msg) => logger.info(msg.trim()) },
   199→      }));
   200→    }
   201→
   202→    // Payment routes (Supabase JWT auth only - no API key needed)
   203→    this.app.use('/api/payments', supabaseAuth, paymentRoutes);
   204→
   205→    // Engine-cut routes (Supabase JWT auth only - no API key needed)
   206→    // Monté AVANT apiKeyAuth car l'app Flutter envoie un JWT, pas de X-API-Key.
   207→    // Le supabaseAuth est appliqué au niveau de la route dans engineCutRoutes.js,
   208→    // pas au niveau du app.use — donc les autres routes /api/devices/* continuent
   209→    // d'utiliser apiKeyAuth via routes.js.
   210→    this.app.use('/api/devices', engineCutRoutes);
   211→
   212→    // Navigation proxy routes (Supabase JWT auth only)
   213→    // Proxy vers Mapbox Directions API + compteur d'usage en DB.
   214→    // Monté AVANT apiKeyAuth car l'app Flutter envoie un JWT, pas de X-API-Key.
   215→    // Routes :
   216→    //   POST /api/navigation/route  → proxy vers Mapbox Directions
   217→    //   GET  /api/navigation/quota  → conso du mois en cours pour l'user
   218→    this.app.use('/api/navigation', navigationRoutes);
   219→
   220→    // Parking routes (Supabase JWT auth only)
   221→    // Active/désactive le mode parking en 1 seul appel réseau (transaction atomique).
   222→    // Routes :
   223→    //   POST /api/parking/activate   → active parking + crée session
   224→    //   POST /api/parking/deactivate → désactive parking + ferme session
   225→    this.app.use('/api/parking', parkingRoutes);
   226→
   227→    // Auth SMS OTP webhook (Supabase Auth calls this to deliver OTP codes via Infobip)
   228→    // Mounted BEFORE apiKeyAuth because Supabase uses Bearer token, not X-API-Key.
   229→    // The route enforces its own Bearer secret (OTP_WEBHOOK_SECRET) + rate limiting.
   230→    const otpLimiter = rateLimit({
   231→      windowMs: 60 * 1000,
   232→      max: 10,
   233→      standardHeaders: true,
   234→      legacyHeaders: false,
   235→      message: { error: 'OTP webhook rate limit exceeded' },
   236→    });
   237→    this.app.use('/api/auth', otpLimiter, authRoutes);
   238→
   239→    // Webhook routes (NO auth - external services call these)
   240→    this.app.use('/webhooks', webhookRoutes);
   241→
   242→    // API Key authentication (for all other /api routes)
   243→    this.app.use('/api', apiKeyAuth);
   244→
   245→    // Routes
   246→    this.app.use('/api', routes);
   247→
   248→    // Root endpoint
   249→    this.app.get('/', (req, res) => {
   250→      res.json({
   251→        name: 'YAZZ GPS Tracking Backend',
   252→        version: '1.0.0',
   253→        status: 'running',
   254→        tcpPort: config.tcpPort,
   255→        httpPort: config.httpPort,
   256→        endpoints: {
   257→          health: '/api/health',
   258→          stats: '/api/stats',
   259→          devices: '/api/devices',
   260→          positions: '/api/positions',
   261→          geofences: '/api/geofences',
   262→        },
   263→      });
   264→    });
   265→
   266→    // 404 handler
   267→    this.app.use((req, res) => {
   268→      res.status(404).json({ error: 'Not found' });
   269→    });
   270→
   271→    // Custom error handler
   272→    this.app.use((err, req, res, next) => {
   273→      logger.error('Unhandled error:', err);
   274→      // Report to Sentry if initialized
   275→      if (config.sentry.dsn) {
   276→        Sentry.captureException(err);
   277→      }
   278→      res.status(500).json({ error: 'Internal server error' });
   279→    });
   280→  }

   282→  /**
   283→   * Handle incoming GPS position from TCP server

   285→  async _onPosition(position, socketId) {
   286→    // FIX Bug B : capturer l'état _disconnectNotified AVANT de le supprimer,
   287→    // pour que _checkBackOnline (appelé plus loin dans positionHandler.handle)
   288→    // puisse savoir si le device était offline avant cette frame.
   289→    const wasOfflineNotified = position.imei ? this._disconnectNotified.has(position.imei) : false;
   290→
   291→    if (position.imei) {
   292→      // Device reconnected — clear from offline-tracking Set
   293→      this._disconnectNotified.delete(position.imei);
   294→
   295→      // Cancel any pending disconnect debounce timer.
   296→      // Si une frame arrive, le device est reconnecté → annuler la notif "offline" en attente.
   297→      this._cancelDisconnectDebounce(position.imei);
   298→    }
   299→
   300→    try {
   301→      // Passer wasOfflineNotified au positionHandler pour qu'il déclenche
   302→      // la notif "back online" si besoin (sans dépendre du callback qui lirait
   303→      // un Set déjà vidé).
   304→      await this.positionHandler.handle(position, socketId, { wasOfflineNotified });
   305→    } catch (err) {
   306→      logger.error(`Error handling position from ${socketId}:`, err);
   307→    }
   308→  }

   317→  /**
   318→   * Handle device TCP disconnection.
   319→   *
   320→   * Strategy: mark is_connected = false in DB ONLY (Flutter sees "offline" instantly
   321→   * via Supabase Realtime). No notification sent here — 90% of TCP disconnects are
   322→   * micro-coupures (antenna switch, tunnel) and would cause spam.
   323→   *
   324→   * Real disconnection notification is delegated to _cleanupStaleConnections()
   325→   * which confirms offline status after 5 min without data and sends via sendAlert()
   326→   * (push + SMS + WhatsApp, multi-user, with credit gate).
   327→   *
   328→   * IMPORTANT: This method does NOT handle acc_on/acc_off (alimentation) — those are
   329→   * evaluated in alertEngine.js from GPS position data (accStatus field), completely
   330→   * independent from TCP socket state.
   331→   */
   332→  async _onDeviceDisconnect(socketId, imei) {
   333→    if (!imei) return;
   334→    logger.info(`[DISCONNECT] Device ${imei} disconnected (socket ${socketId})`);
   335→    try {
   336→      const supabase = getSupabaseService();
   337→      await supabase.markDeviceDisconnected(imei);
   338→
   339→      // ═══════════════════════════════════════════════════════════════
   340→      // ADAPTATIVE DEBOUNCE — Timer selon le contexte véhicule
   341→      // ═══════════════════════════════════════════════════════════════
   342→      // Logique (proposition user + anti-fausse-alerte parking) :
   343→      //
   344→      // 1. Véhicule EN MARCHE (acc_status=true) → 5 min
   345→      //    Probabilité : tunnel, pont, zone blanche → on attend pour éviter spam
   346→      //
   347→      // 2. Véhicule ARRÊTÉ DEPUIS > 5 min (acc_status=false, last_update vieux)
   348→      //    Probabilité : sabotage, batterie morte → on alerte vite (30s)
   349→      //
   350→      // 3. Véhicule ARRÊTÉ RÉCEMMENT (< 5 min après acc_status→false)
   351→      //    Probabilité : stationnement normal, l'utilisateur vient de garer
   352→      //    → on attend 5 min pour éviter fausse alerte au parking
   353→      // ═══════════════════════════════════════════════════════════════
   354→
   355→      // Annuler un timer existant pour cet IMEI
   356→      if (this._disconnectTimers?.has(imei)) {
   357→        clearTimeout(this._disconnectTimers.get(imei));
   358→      }
   359→      if (!this._disconnectTimers) this._disconnectTimers = new Map();
   360→
   361→      // Récupérer l'état ACC et last_update pour déterminer le délai adaptatif
   362→      const { data: currentPos } = await supabase.client
   363→        .from('last_known_positions')
   364→        .select('acc_status, last_update')
   365→        .eq('device_id', imei)
   366→        .maybeSingle();
   367→
   368→      const accStatus = currentPos?.acc_status;
   369→      const lastUpdateMs = currentPos?.last_update
   370→        ? Date.now() - new Date(currentPos.last_update).getTime()
   371→        : null;
   372→
   373→      // Déterminer le délai selon l'état du véhicule (logique adaptative)
   374→      let debounceDelay;
   375→      let delayReason;
   376→
   377→      if (accStatus === true) {
   378→        // Véhicule en marche → tunnel/pont probable → 5 min
   379→        debounceDelay = 5 * 60 * 1000;
   380→        delayReason = 'vehicle_running_5min';
   381→      } else if (accStatus === false && lastUpdateMs !== null && lastUpdateMs > 5 * 60 * 1000) {
   382→        // Véhicule arrêté depuis > 5 min → sabotage probable → 30s
   383→        debounceDelay = 30 * 1000;
   384→        delayReason = 'vehicle_parked_long_30s';
   385→      } else {
   386→        // Véhicule arrêté récemment (< 5 min) ou état inconnu → stationnement normal → 5 min
   387→        debounceDelay = 5 * 60 * 1000;
   388→        delayReason = 'vehicle_parked_recent_5min';
   389→      }
   390→
   391→      logger.info(`[DISCONNECT] ${imei}: marked offline, ${delayReason} timer started (${debounceDelay / 1000}s) — acc=${accStatus}, lastUpdateAge=${lastUpdateMs ? Math.round(lastUpdateMs / 1000) + 's' : 'null'}`);
   392→
   393→      // Démarrer le timer adaptatif
   394→      const timer = setTimeout(async () => {
   395→        try {
   396→          // Double-check device is still offline (didn't reconnect during 30s)
   397→          const { data: pos } = await supabase.client
   398→            .from('last_known_positions')
   399→            .select('is_connected, last_update')
   400→            .eq('device_id', imei)
   401→            .maybeSingle();
   402→
   403→          if (pos?.is_connected === true) {
   404→            // Device reconnected during the 30s window — skip notification
   405→            logger.info(`[DISCONNECT] ${imei}: reconnected within 30s debounce, skipping offline notification`);
   406→            this._disconnectTimers.delete(imei);
   407→            return;
   408→          }
   409→
   410→          // Check if we already notified (anti-spam)
   411→          if (this._disconnectNotified.has(imei)) {
   412→            logger.debug(`[DISCONNECT] ${imei}: already notified, skipping`);
   413→            this._disconnectTimers.delete(imei);
   414→            return;
   415→          }
   416→
   417→          // Send HIGH PRIORITY notification
   418→          const deviceName = await supabase.getDeviceName(imei) || imei;
   419→          const allUsers = await supabase.getDeviceAllUsers(imei);
   420→
   421→          if (!allUsers || allUsers.length === 0) {
   422→            logger.info(`[DISCONNECT] ${imei}: no linked users, skipping notification`);
   423→            this._disconnectTimers.delete(imei);
   424→            return;
   425→          }
   426→
   427→          const notificationService = getNotificationService();
   428→          const result = await notificationService.sendAlert({
   429→            users: allUsers.map(u => ({ userId: u.userId })),
   430→            type: 'device_offline',
   431→            title: '🚨 Capteur déconnecté',
   432→            body: `${deviceName} est hors ligne — Vérifiez votre véhicule`,
   433→            deviceId: imei,
   434→            data: {
   435→              type: 'device_offline',
   436→              device_id: imei,
   437→              source: 'tcp_disconnect_debounced',
   438→              priority: 'critical',
   439→            },
   440→            priority: 'critical', // FIX #3: signal to notificationService to use MAX priority
   441→          });
   442→
   443→          // Mark as notified
   444→          this._disconnectNotified.add(imei);
   445→
   446→          const channelSummary = result
   447→            .map(r => {
   448→              const chans = (r.results || []).map(cr => `${cr.channel}=${cr.status}`).join(',');
   449→              return `user=${r.userId.substring(0, 8)} [${chans}]`;
   450→            })
   451→            .join(' | ');
   452→          logger.info(`[DISCONNECT] ${imei}: sent offline notification (${delayReason}, ${debounceDelay / 1000}s debounce) → ${channelSummary}`);
   453→        } catch (err) {
   454→          logger.error(`[DISCONNECT] ${imei}: debounced notification failed:`, err.message);
   455→        } finally {
   456→          this._disconnectTimers.delete(imei);
   457→        }
   458→      }, debounceDelay);
   459→
   460→      this._disconnectTimers.set(imei, timer);
   461→
   462→      // Clear back-online check cache so next reconnect triggers fresh check
   463→      if (this.positionHandler) {
   464→        this.positionHandler.clearBackOnlineCheck(imei);
   465→      }
   466→    } catch (err) {
   467→      logger.error(`[DISCONNECT] Failed for ${imei}:`, err);
   468→    }
   469→  }

   471→  /**
   472→   * Cancel the disconnect debounce timer when a position arrives (device reconnected).
   473→   * Called from _onPosition() in server.js.
   474→   */
   475→  _cancelDisconnectDebounce(imei) {
   476→    if (this._disconnectTimers?.has(imei)) {
   477→      clearTimeout(this._disconnectTimers.get(imei));
   478→      this._disconnectTimers.delete(imei);
   479→      logger.info(`[RECONNECT] ${imei}: cancelled pending disconnect notification (micro-coupure)`);
   480→      return true;
   481→    }
   482→    return false;
   483→  }

   485→  /**
   486→   * Setup periodic cleanup tasks
   487→   * - Alert engine dedup cleanup (existing — unchanged)
   488→   * - Alert engine device state cleanup (existing — unchanged)
   489→   * - Stale connection cleanup: detects devices offline > 5 min, sends
   490→   *   notification via sendAlert() (push + SMS + WhatsApp) to all linked users
   491→   */
   492→  _setupCleanup() {
   493→    // Run cleanup every 2 minutes
   494→    this._cleanupInterval = setInterval(() => {
   495→      try {
   496→        const alertEngine = getAlertEngine();
   497→        alertEngine.cleanupDedup();
   498→        alertEngine.cleanupDeviceStates();
   499→
   500→        // Detect devices with no data for 5+ min, notify via all channels
   501→        this._cleanupStaleConnections();
   502→
   503→        logger.info('Periodic cleanup completed');
   504→      } catch (err) {
   505→        logger.error('Cleanup failed:', err);
   506→      }
   507→    }, 2 * 60 * 1000); // Every 2 minutes
   508→  }
   509→
   510→  /**
   511→   * Detect devices truly offline (no data for 5+ min) and send notification
   512→   * via the unified sendAlert() — covers push + SMS + WhatsApp for all linked users.
   513→   *
   514→   * Covers two scenarios:
   515→   * 1. TCP close fired but notification was deferred (see _onDeviceDisconnect)
   516→   * 2. TCP close never fired (silent network drop — e.g. device powered off)
   517→   *
   518→   * Anti-spam: _disconnectNotified Set prevents re-notifying the same IMEI
   519→   * while it remains offline. Cleared instantly on reconnection in _onPosition().
   520→   *
   521→   * NOTE: This only handles device_offline (network disconnect). It does NOT
   522→   * handle acc_on/acc_off (alimentation) — those are in alertEngine.js.
   523→   */
   524→  async _cleanupStaleConnections() {
   525→    try {
   526→      const supabase = getSupabaseService();
   527→      const notificationService = getNotificationService();
   528→
   529→      // Find devices still marked connected but with no data for 5 minutes
   530→      const cutoff = new Date(Date.now() - 5 * 60 * 1000).toISOString();
   531→      const { data: staleDevices, error } = await supabase.client
   532→        .from('last_known_positions')
   533→        .select('device_id')
   534→        .eq('is_connected', true)
   535→        .lt('last_update', cutoff);
   536→
   537→      if (error || !staleDevices || staleDevices.length === 0) {
   538→        return; // No stale devices — silent exit (no log spam)
   539→      }
   540→
   541→      // Mark them all disconnected in DB
   542→      const staleImeis = staleDevices.map(d => d.device_id);
   543→      const { error: updateErr } = await supabase.client
   544→        .from('last_known_positions')
   545→        .update({ is_connected: false })
   546→        .in('device_id', staleImeis);
   547→
   548→      if (updateErr) {
   549→        logger.error('Failed to mark stale devices disconnected:', updateErr);
   550→        return;
   551→      }
   552→
   553→      logger.info(`🧹 Stale cleanup: ${staleDevices.length} device(s) offline > 5min: ${staleImeis.join(', ')}`);
   554→
   555→      // Send notification via unified sendAlert() for each non-duplicate device
   556→      for (const { device_id: imei } of staleDevices) {
   557→        // Anti-spam: skip if already notified for this offline period
   558→        if (this._disconnectNotified.has(imei)) {
   559→          logger.debug(`[STALE] ${imei}: device_offline notification SKIPPED (already notified, awaiting reconnection)`);
   560→          continue;
   561→        }
   562→
   563→        try {
   564→          // Get device name for human-readable notification body
   565→          const deviceName = await supabase.getDeviceName(imei) || imei;
   566→
   567→          // Get ALL users linked to this device (owner + shared users)
   568→          const allUsers = await supabase.getDeviceAllUsers(imei);
   569→          if (!allUsers || allUsers.length === 0) {
   570→            logger.info(`[STALE] ${imei}: no linked users found, skipping notification`);
   571→            continue;
   572→          }
   573→
   574→          // Send via unified sendAlert() — handles channel selection, credit gate,
   575→          // DB insert, push + SMS + WhatsApp per user preference
   576→          const result = await notificationService.sendAlert({
   577→            users: allUsers.map(u => ({ userId: u.userId })),
   578→            type: 'device_offline',
   579→            title: 'Capteur déconnecté',
   580→            body: `${deviceName} est hors ligne`,
   581→            deviceId: imei,
   582→            data: { type: 'device_offline', device_id: imei, source: 'stale_cleanup' },
   583→          });
   584→
   585→          // Mark as notified to prevent re-sending while device stays offline
   586→          this._disconnectNotified.add(imei);
   587→
   588→          const channelSummary = result
   589→            .map(r => {
   590→              const chans = (r.results || []).map(cr => `${cr.channel}=${cr.status}`).join(',');
   591→              return `user=${r.userId.substring(0, 8)} [${chans}]`;
   592→            })
   593→            .join(' | ');
   594→          logger.info(`[STALE] ${imei}: sendAlert result → ${channelSummary}`);
   595→        } catch (err) {
   596→          logger.error(`[STALE] Notification failed for ${imei}:`, err.message);
   597→          // Don't add to _disconnectNotified on error — retry on next cycle
   598→        }
   599→      }
   600→    } catch (err) {
   601→      logger.error('Stale connection cleanup failed:', err);
   602→    }
   603→  }
   604→
   605→  /**
   606→   * Setup graceful shutdown handlers
   607→   */
   608→  _setupShutdown() {
   609→    const shutdown = async (signal) => {
   610→      if (this._shuttingDown) return;
   611→      this._shuttingDown = true;
   612→
   613→      logger.info(`Received ${signal}, shutting down gracefully...`);
   614→
   615→      // Clear cleanup interval
   616→      if (this._cleanupInterval) {
   617→        clearInterval(this._cleanupInterval);
   618→      }
   619→
   620→      // Stop Realtime listener
   621→      const notificationService = getNotificationService();
   622→      notificationService.stopRealtimeListener();
   623→
   624→      // Stop Parking Alert Loops (bip-bip critique)
   625→      // Important : arrête les boucles proprement au shutdown pour éviter
   626→      // les résidus au redémarrage.
   627→      try {
   628→        const { getParkingAlertLoop } = require('./services/parkingAlertLoop');
   629→        getParkingAlertLoop().stopAll();
   630→      } catch (e) {
   631→        logger.warn('ParkingAlertLoop stop failed (non-critical):', e.message);
   632→      }
   633→
   634→      // Stop TCP server
   635→      if (this.tcpServer) {
   636→        await this.tcpServer.stop();
   637→      }
   638→
   639→      // Stop HTTP server
   640→      if (this.httpServer) {
   641→        this.httpServer.close(() => {
   642→          logger.info('HTTP server closed');
   643→          process.exit(0);
   644→        });
   645→
   646→        // Force exit after 10 seconds
   647→        setTimeout(() => {
   648→          logger.error('Forced shutdown after timeout');
   649→          process.exit(1);
   650→        }, 10000);
   651→      } else {
   652→        process.exit(0);
   653→      }
   654→    };
   655→
   656→    process.on('SIGTERM', () => shutdown('SIGTERM'));
   657→    process.on('SIGINT', () => shutdown('SIGINT'));
   658→    process.on('SIGQUIT', () => shutdown('SIGQUIT'));
   659→
   660→    // Handle uncaught errors
   661→    process.on('uncaughtException', (err) => {
   662→      logger.error('Uncaught exception:', err);
   663→      shutdown('uncaughtException');
   664→    });
   665→
   666→    process.on('unhandledRejection', (reason) => {
   667→      logger.error('Unhandled rejection:', reason);
   668→      shutdown('unhandledRejection');
   669→    });
   670→  }
   671→}
