     1→const logger = require('../utils/logger');
     2→const { config } = require('../utils/config');

     6→/**
     7→ * Notification Service
     8→ *
     9→ * Handles sending notifications through multiple channels:
    10→ * - Push: Firebase Cloud Messaging (FCM) — FREE
    11→ * - WhatsApp: Provider-based (Infobip / Twilio) — PAID (deducted from user credits)
    12→ * - SMS: Provider-based (Infobip / Twilio) — PAID (deducted from user credits)
    13→ *
    14→ * Provider selection is dynamic:
    15→ * - Configured via app_settings.notification_provider in Supabase (switchable from admin panel)
    16→ * - Fallback: NOTIFICATION_PROVIDER env var, then first available provider
    17→ * - Cache: 60 seconds (avoids DB query on every notification)
    18→ *
    19→ * Paid channels (whatsapp/sms) deduct from user credit balance after successful send.
    20→ * Also includes a Supabase Realtime listener for pg_cron notifications.
    21→ */
    22→
    23→class NotificationService {
    24→  constructor() {
    25→    this.supabase = getSupabaseService();
    26→    this.fcmInitialized = false;
    27→    this.messaging = null;
    28→    this._realtimeChannel = null;
    29→
    30→    // Multi-provider system
    31→    this.providers = {};
    32→    this._activeProviderCache = null;
    33→    this._activeProviderCacheTime = 0;
    34→
    35→    this._initFCM();
    36→    this._initProviders();
    37→  }

    43→  _initFCM() {
    44→    try {
    45→      const admin = require('firebase-admin');
    46→      let credential;
    47→
    48→      if (config.firebase.projectId && config.firebase.privateKey && config.firebase.clientEmail) {
    49→        credential = admin.credential.cert({
    50→          projectId: config.firebase.projectId,
    51→          privateKey: config.firebase.privateKey,
    52→          clientEmail: config.firebase.clientEmail,
    53→        });
    54→        logger.info('Firebase: using env var credentials');
    55→      } else {
    56→        const fs = require('fs');
    57→        const path = require('path');
    58→        const serviceAccountPath = path.join(__dirname, '../../firebase-service-account.json');
    59→
    60→        if (fs.existsSync(serviceAccountPath)) {
    61→          const serviceAccount = JSON.parse(fs.readFileSync(serviceAccountPath, 'utf8'));
    62→          credential = admin.credential.cert(serviceAccount);
    63→          logger.info('Firebase: using service account JSON file');
    64→        } else {
    65→          logger.warn('Firebase credentials not configured (no env vars, no JSON file) — FCM notifications disabled');
    66→          return;
    67→        }
    68→      }
    69→
    70→      admin.initializeApp({ credential });
    71→
    72→      this.messaging = admin.messaging();
    73→      this.fcmInitialized = true;
    74→      logger.info('Firebase Admin SDK initialized — FCM notifications enabled');
    75→    } catch (err) {
    76→      logger.error('Failed to initialize Firebase Admin SDK:', err);
    77→      this.fcmInitialized = false;
    78→    }
    79→  }

    81→  /**
    82→   * Initialize all available notification providers
    83→   * Each provider self-validates its credentials and only activates if configured
    84→   */
    85→  _initProviders() {
    86→    try {
    87→      // Initialize Infobip provider
    88→      const InfobipProvider = require('./providers/infobipProvider');
    89→      const infobip = new InfobipProvider();
    90→      infobip.init().then(() => {
    91→        if (infobip.getStatus().initialized) {
    92→          this.providers.infobip = infobip;
    93→          logger.info('✅ Infobip provider registered');
    94→        }
    95→      }).catch(err => {
    96→        logger.warn('Infobip provider init failed:', err.message);
    97→      });
    98→
    99→      // Initialize Twilio provider
   100→      const TwilioProvider = require('./providers/twilioProvider');
   101→      const twilio = new TwilioProvider();
   102→      twilio.init().then(() => {
   103→        if (twilio.getStatus().initialized) {
   104→          this.providers.twilio = twilio;
   105→          logger.info('✅ Twilio provider registered');
   106→        }
   107→      }).catch(err => {
   108→        logger.warn('Twilio provider init failed:', err.message);
   109→      });
   110→
   111→      // Initialize HTTPSMS provider (SMS only)
   112→      const HttpsmsProvider = require('./providers/httpsmsProvider');
   113→      const httpsms = new HttpsmsProvider();
   114→      httpsms.init().then(() => {
   115→        if (httpsms.getStatus().initialized) {
   116→          this.providers.httpsms = httpsms;
   117→          logger.info('✅ HTTPSMS provider registered');
   118→        }
   119→      }).catch(err => {
   120→        logger.warn('HTTPSMS provider init failed:', err.message);
   121→      });
   122→    } catch (err) {
   123→      logger.error('Failed to initialize notification providers:', err);
   124→    }

   127→  /**
   128→   * Get the active notification provider
   129→   * Reads from app_settings.notification_provider (cached 60s)
   130→   * Falls back to NOTIFICATION_PROVIDER env var, then first available provider
   131→   */
   132→  async getActiveProvider() {
   133→    const now = Date.now();
   134→    if (this._activeProviderCache && now - this._activeProviderCacheTime < 60000) {
   135→      return this._activeProviderCache;
   136→    }
   137→
   138→    try {
   139→      const { data } = await this.supabase.client
   140→        .from('app_settings')
   141→        .select('value')
   142→        .eq('key', 'notification_provider')
   143→        .single();
   144→
   145→      const providerName = data?.value || config.notificationProvider || 'infobip';
   146→      const provider = this.providers[providerName];
   147→
   148→      if (provider) {
   149→        this._activeProviderCache = provider;
   150→      } else {
   151→        // Fallback: first available provider
   152→        const available = Object.keys(this.providers);
   153→        if (available.length > 0) {
   154→          logger.warn(`Provider "${providerName}" not available, falling back to "${available[0]}"`);
   155→          this._activeProviderCache = this.providers[available[0]];
   156→        } else {
   157→          logger.error('No notification provider available!');
   158→          this._activeProviderCache = null;
   159→        }
   160→      }
   161→    } catch (err) {
   162→      logger.error('Failed to read notification_provider from app_settings:', err.message);
   163→      const fallbackName = config.notificationProvider || 'infobip';
   164→      this._activeProviderCache = this.providers[fallbackName] || Object.values(this.providers)[0] || null;
   165→    }
   166→
   167→    this._activeProviderCacheTime = now;
   168→    return this._activeProviderCache;
   169→  }

   190→  // AUDIT ANTI-SPAM — Garde-fou global contre les rafales de notifications.
   191→  // Cache mémoire : clé = "userId:deviceId:alertType" → timestamp du dernier envoi.
   192→  // Empêche les doublons ET les rafales (ex: 4 frames en 38ms → 1 seule notif).
   193→  static COOLDOWN_MS = {
   194→    gps_frozen: 24 * 60 * 60 * 1000,         // 24 heures
   195→    tow: 60 * 60 * 1000,                     // 1 heure
   196→    gps_jamming: 60 * 60 * 1000,             // 1 heure
   197→    gsm_jamming: 60 * 60 * 1000,             // 1 heure
   198→    gps_no_jamming: 60 * 60 * 1000,          // 1 heure
   199→    gsm_no_jamming: 60 * 60 * 1000,          // 1 heure
   200→    battery_low: 60 * 60 * 1000,             // 1 heure
   201→    parking_movement: 60 * 60 * 1000,        // 1 heure
   202→    parking_voltage: 60 * 60 * 1000,         // 1 heure
   203→    parking_ignition: 60 * 60 * 1000,        // 1 heure
   204→    power_off: 10 * 60 * 1000,               // 10 minutes
   205→    device_offline: 10 * 60 * 1000,          // 10 minutes
   206→    voltage_drop: 60 * 60 * 1000,            // 1 heure
   207→    engine_cut: 5 * 60 * 1000,               // 5 minutes
   208→    engine_restore: 5 * 60 * 1000,           // 5 minutes
   209→  };
   210→  static DEFAULT_COOLDOWN_MS = 5 * 60 * 1000; // 5 minutes par défaut
   211→
   212→  _globalNotifCooldown = new Map();
   213→
   214→  /**
   215→   * Vérifie si une notification peut être envoyée (pas en cooldown).
   216→   * @returns {boolean} true si l'envoi est autorisé, false si bloqué par le cooldown.
   217→   */
   218→  _isNotifSpamming(userId, deviceId, alertType) {
   219→    const key = `${userId}:${deviceId}:${alertType}`;
   220→    const now = Date.now();
   221→    const cooldownMs = NotificationService.COOLDOWN_MS[alertType] || NotificationService.DEFAULT_COOLDOWN_MS;
   222→    
   223→    const lastSent = this._globalNotifCooldown.get(key);
   224→    if (lastSent && (now - lastSent) < cooldownMs) {
   225→      logger.info(`🚫 ANTI-SPAM: blocked ${alertType} for user ${userId?.substring(0,8)} device ${deviceId?.substring(0,8)} (cooldown ${Math.round(cooldownMs/60000)}min)`);
   226→      return true; // Spamming — bloquer
   227→    }
   228→    
   229→    // Pas en cooldown — autoriser et marquer le timestamp
   230→    this._globalNotifCooldown.set(key, now);
   231→    return false;
   232→  }

   234→  async sendAlert(alert) {
   235→    const creditService = getCreditService();
   236→    const results = [];
   237→
   238→    // FIX (anti-doublons notifications) : Dédoublonnage par token FCM.
   239→    //
   240→    // Problème : Si plusieurs comptes users partagent le MÊME token FCM (cas
   241→    // typique : comptes de test connectés sur le même téléphone, ou utilisateur
   242→    // qui change de compte sans logout propre), le backend envoie 1 push par
   243→    // user → le téléphone reçoit N pushes pour le même event → notifications
   244→    // dédoublonnées en barre de tâche.
   245→    //
   246→    // Solution : On track les tokens FCM déjà traités dans cette boucle, et on
   247→    // skip les users suivants qui ont le même token. Le 1er user rencontré
   248→    // (l'owner en général, car getDeviceAllUsers retourne owners en premier)
   249→    // "gagne" le push. Les autres users avec le même token sont ignorés
   250→    // silencieusement (log de debug pour traçabilité).
   251→    //
   252→    // Note : La row en DB est quand même créée pour CHAQUE user (avant le skip),
   253→    // pour que la page notifications de l'app affiche l'alerte pour chaque
   254→    // compte individuellement. Seul l'envoi FCM est dédoublonné.
   255→    const seenFcmTokens = new Set();
   256→
   257→    for (const user of alert.users) {
   258→      try {
   259→        // AUDIT ANTI-SPAM — Garde-fou global : vérifier le cooldown AVANT tout
   260→        if (this._isNotifSpamming(user.userId, alert.deviceId, alert.type)) {
   261→          continue; // Skip cette notification (spam bloqué)
   262→        }
   263→
   264→        const userDetails = await this.supabase.getUser(user.userId);
   265→        if (!userDetails) {
   266→          logger.warn(`User ${user.userId} not found, skipping notification`);
   267→          continue;
   268→        }
   269→
   270→        // FIX (anti-doublons) : Detecter si ce token FCM a déjà été vu pour
   271→        // un autre user dans cette même alerte. Si oui, on skip l'envoi push
   272→        // (mais on crée quand même la row DB plus bas pour la page notif).
   273→        // Comportement symétrique pour owner et receiver : le 1er user
   274→        // rencontré (l'owner en général, car getDeviceAllUsers retourne
   275→        // owners en premier) gagne le push, les suivants avec le même token
   276→        // sont skippés.
   277→        let fcmTokenAlreadyUsed = false;
   278→        if (userDetails.fcm_token) {
   279→          if (seenFcmTokens.has(userDetails.fcm_token)) {
   280→            fcmTokenAlreadyUsed = true;
   281→            logger.debug(`🔁 User ${user.userId} shares FCM token with another user — push will be skipped, DB row still created`);
   282→          } else {
   283→            seenFcmTokens.add(userDetails.fcm_token);
   284→          }
   285→        }
   286→
   287→        if (!this._isAlertEnabled(alert.type, userDetails)) {
   288→          logger.debug(`Alert type ${alert.type} disabled for user ${user.userId}`);
   289→          continue;
   290→        }
   291→
   292→        // ── Credit gate: skip PAID channels only if credits depleted ──
   293→        // Push is always FREE and must never be blocked.
   294→        // Exception: credit_reminder and credit_depleted must always go through.
   295→        const isCreditNotification = alert.type === 'credit_reminder' || alert.type === 'credit_depleted';
   296→        let creditsDepleted = false;
   297→        if (!isCreditNotification) {
   298→          const credits = await creditService.getBalance(user.userId);
   299→          creditsDepleted = credits && !credits.is_active;
   300→        }
   301→
   302→        // V3 logic: Push is ALWAYS the base (free). User picks AT MOST ONE paid supplement.
   303→        const paidSupplement = (userDetails.paid_supplement && userDetails.paid_supplement !== 'none')
   304→          ? userDetails.paid_supplement
   305→          : (userDetails.alert_channel === 'sms' || userDetails.alert_channel === 'whatsapp')
   306→            ? userDetails.alert_channel
   307→            : 'none';
   308→        let channels;
   309→        switch (paidSupplement) {
   310→          case 'sms':
   311→            channels = ['push', 'sms'];
   312→            break;
   313→          case 'whatsapp':
   314→            channels = ['push', 'whatsapp'];
   315→            break;
   316→          default:
   317→            channels = ['push'];
   318→            const channelPref = userDetails.alert_channel || 'push';
   319→            if (channelPref === 'all' || channelPref === 'whatsapp_sms' || channelPref === 'whatsapp+sms') {
   320→              channels = ['push', 'sms'];
   321→            }
   322→        }
   323→
   324→        const sendResults = [];
   325→
   326→        for (const ch of channels) {
   327→          if (ch === 'whatsapp' || ch === 'sms') {
   328→            const creditCheck = await creditService.checkNotificationCredit(user.userId, ch);
   329→            if (!creditCheck.allowed) {
   330→              logger.warn(`Insufficient credits for ${ch} notification to user ${user.userId} (balance: ${creditCheck.balance}, fee: ${creditCheck.fee})`);
   331→              sendResults.push({
   332→                channel: ch,
   333→                status: 'skipped',
   334→                reason: `Insufficient credits (need ${creditCheck.fee} CDF, balance: ${creditCheck.balance})`,
   335→              });
   336→              continue;
   337→            }
   338→          }
   339→
   340→          let result;
   341→          switch (ch) {
   342→            case 'push':
   343→              // FIX (anti-doublons) : Si le token FCM de ce user a déjà été
   344→              // utilisé pour cette alerte (un autre user avec le même token a
   345→              // déjà reçu le push), on skip l'envoi FCM mais on marque le
   346→              // résultat comme 'skipped_duplicate' pour traçabilité.
   347→              // La row DB sera quand même créée plus bas (pour la page notif
   348→              // de l'app), c'est seulement l'envoi physique du push qui est
   349→              // dédoublonné.
   350→              if (userDetails.fcm_token && fcmTokenAlreadyUsed) {
   351→                result = {
   352→                  channel: 'push',
   353→                  status: 'skipped_duplicate',
   354→                  reason: 'FCM token already used by another user in this alert batch',
   355→                  userId: user.userId,
   356→                };
   357→                logger.debug(`🔁 Push skipped for user ${user.userId} (duplicate FCM token)`);
   358→              } else {
   359→                result = await this._sendPush(userDetails, alert);
   360→              }
   361→              break;
   362→            case 'whatsapp':
   363→              result = await this._sendWhatsApp(userDetails, alert);
   364→              break;
   365→            case 'sms':
   366→              result = await this._sendSMS(userDetails, alert);
   367→              break;
   368→          }
   369→
   370→          if ((ch === 'whatsapp' || ch === 'sms') && result.status === 'sent') {
   371→            const feeType = ch === 'whatsapp' ? 'whatsapp_fee' : 'sms_fee';
   372→            const deduction = await creditService.deductNotificationFee(user.userId, feeType);
   373→            if (deduction.success) {
   374→              result.deduction = { fee: deduction.fee, newBalance: deduction.newBalance };
   375→              try {
   376→                await this.supabase.client.from('notification_fee_log').insert({
   377→                  user_id: user.userId,
   378→                  notification_id: result.notificationId || null,
   379→                  channel: ch,
   380→                  fee_amount: deduction.fee,
   381→                  balance_before: deduction.newBalance + deduction.fee,
   382→                  balance_after: deduction.newBalance,
   383→                });
   384→              } catch (logErr) {
   385→                logger.warn('Failed to log notification fee:', logErr.message);
   386→              }
   387→            }
   388→          }
   389→
   390→          sendResults.push(result);
   391→        }
   392→
   393→        const persisted = await this.supabase.insertNotification({
   394→          userId: user.userId,
   395→          deviceId: alert.deviceId,
   396→          type: alert.type,
   397→          title: alert.title,
   398→          body: alert.body,
   399→          // FIX: Ajouter direct_send: true pour que le Realtime listener
   400→          // sache que le FCM a déjà été envoyé directement par sendAlert()
   401→          // et ne le renvoie pas une 2ème fois (doublon FCM).
   402→          data: { ...alert.data, direct_send: true },
   403→          channel: channels.join(','),
   404→        });
   405→
   406→        results.push({
   407→          userId: user.userId,
   408→          channels: channels,
   409→          results: sendResults,
   410→          persisted: !!persisted,
   411→        });
   412→      } catch (err) {
   413→        logger.error(`Failed to send notification to user ${user.userId}:`, err);
   414→        results.push({ userId: user.userId, error: err.message });
   415→      }
   416→    }
   417→
   418→    return results;
   419→  }

   425→  async _sendPush(user, alert) {
   426→    if (!this.fcmInitialized) {
   427→      return { channel: 'push', status: 'skipped', reason: 'FCM not configured' };
   428→    }
   429→    if (!user.fcm_token) {
   430→      return { channel: 'push', status: 'skipped', reason: 'No FCM token' };
   431→    }
   432→
   433→    try {
   434→      // FIX #3: Critical alerts get PRIORITY_MAX + heads-up + vibration pattern.
   435→      // Non-critical alerts keep the default 'yazz_alerts' channel (normal priority).
   436→      // Critical alerts use 'yazz_alerts_critical' channel (configured in Flutter
   437→      // with IMPORTANCE_HIGH + sound + vibration + heads-up popup).
   438→      //
   439→      // Critical = alerts that matter for vehicle security or that the user
   440→      // explicitly configured as "important" (speed_limit, geofence, etc.).
   441→      // These get FCM priority 'max' to force heads-up display on Android.
   442→      const isCritical = alert.priority === 'critical'
   443→        || alert.type === 'device_offline'
   444→        || alert.type === 'device_online'
   445→        || alert.type === 'sos'
   446→        || alert.type === 'tow'
   447→        || alert.type === 'impact'
   448→        || alert.type === 'power_cut'
   449→        || alert.type === 'power_off'
   450→        || alert.type === 'battery_disconnect'
   451→        || alert.type === 'sensor_disconnect'
   452→        // FIX: speed_limit, geofence, ignition = also critical for security
   453→        || alert.type === 'speed_limit'
   454→        || alert.type === 'geofence_enter'
   455→        || alert.type === 'geofence_exit'
   456→        || alert.type === 'acc_on'
   457→        || alert.type === 'acc_off'
   458→        // ÉTAPE 0 — Parking restauré (alertes antivol critiques)
   459→        || alert.type === 'parking_movement'
   460→        || alert.type === 'parking_voltage'
   461→        || alert.type === 'parking_ignition'
   462→        // ÉTAPE 1 — Brouillage GPS/GSM (alertes antivol critiques)
   463→        || alert.type === 'gps_jamming'
   464→        || alert.type === 'gsm_jamming'
   465→        // ÉTAPE 2 — Coupure moteur (alerte critique quand moteur coupé)
   466→        || alert.type === 'engine_cut'
   467→        // AUDIT 3 — Capteur figé (alerte critique — possible problème d'installation)
   468→        || alert.type === 'gps_frozen';
   469→
   470→      // Phase 2 (redesign parking) : alerte parking_movement en boucle critique.
   471→      // Utilise un channel Android dédié 'critical_parking' avec bypass Dnd +
   472→      // son 'critical_alert.mp3' spécifique (bip-bip 30s).
   473→      // Les autres alertes critiques restent sur 'yazz_alerts_critical' (inchangé).
   474→      const isParkingLoopAlert = alert.type === 'parking_movement'
   475→        && alert.data && alert.data.parking_alert === 'true';
   476→
   477→      // FIX #4: Send as DATA-ONLY message (no `notification` field).
   478→      //
   479→      // Why: When FCM receives a message with a `notification` field, the Android
   480→      // system displays it AUTOMATICALLY using a default channel (ignoring our
   481→      // yazz_alerts_critical channel + alert.mp3 sound). The Flutter code never
   482→      // gets the chance to call showAlert() with our custom channel.
   483→      //
   484→      // By sending data-only:
   485→      //   - Foreground: FirebaseMessaging.onMessage fires → Flutter calls showAlert()
   486→      //     with our custom channel + sound + vibration
   487→      //   - Background: FirebaseMessaging.onBackgroundMessage fires → Flutter
   488→      //     shows a local notification with our custom channel
   489→      //
   490→      // We pass title/body in `data` so Flutter can read them and call showAlert().
   491→      //
   492→      // Trade-off: iOS requires `notification` field for push to display when app
   493→      // is in background. We use `apns.payload.aps.alert` to handle iOS while
   494→      // keeping Android as data-only.
   495→      const message = {
   496→        token: user.fcm_token,
   497→        data: {
   498→          title: alert.title,
   499→          body: alert.body,
   500→          type: alert.type,
   501→          deviceId: alert.deviceId || '',
   502→          priority: isCritical ? 'critical' : 'normal',
   503→          // Convert all alert.data values to strings (FCM data values must be strings)
   504→          ...Object.fromEntries(
   505→            Object.entries(alert.data || {}).map(([k, v]) => [k, String(v)])
   506→          ),
   507→        },
   508→        android: {
   509→          priority: 'high', // Always high — FCM message delivery priority
   510→          // FIX #4: No `notification` field on Android → message is data-only
   511→          // Flutter receives it via onMessage/onBackgroundMessage and calls showAlert()
   512→          //
   513→          // Phase 2 (redesign parking) : si alerte parking en boucle, on ajoute
   514→          // notificationOptions pour forcer le channel 'critical_parking' côté Flutter.
   515→          // Le channel lui-même est créé côté Flutter (Phase 5) avec bypass Dnd.
   516→          ...(isParkingLoopAlert ? {
   517→            notification: {
   518→              title: alert.title,
   519→              body: alert.body,
   520→              sound: 'critical_alert',
   521→              channelId: 'critical_parking',
   522→              priority: 'max',
   523→              visibility: 'public',
   524→              vibrationPattern: [0, 1000, 500, 1000, 500, 1000],
   525→              // bypass Dnd via le channel côté Flutter (IMPORTANCE_HIGH + setBypassDnd)
   526→            },
   527→          } : {}),
   528→        },
   529→        apns: {
   530→          payload: {
   531→            aps: {
   532→              // FIX #4: iOS requires `alert` field for background display
   533→              // We use the alert object so iOS displays the notification natively
   534→              alert: {
   535→                title: alert.title,
   536→                body: alert.body,
   537→              },
   538→              sound: isCritical ? 'critical_alert.caf' : 'alert.mp3',
   539→              badge: 1,
   540→              // Critical alerts bypass Do Not Disturb on iOS
   541→              'content-available': 1,
   542→            },
   543→          },
   544→          headers: isCritical ? {
   545→            'apns-priority': '10', // Highest priority
   546→            'apns-push-type': 'alert',
   547→          } : {},
   548→        },
   549→      };
   550→
   551→      const response = await this.messaging.send(message);
   552→      logger.info(`FCM sent to user ${user.id}: ${response} (priority=${isCritical ? 'critical' : 'normal'}, mode=data-only)`);
   553→      return { channel: 'push', status: 'sent', messageId: response, priority: isCritical ? 'critical' : 'normal' };
   554→    } catch (err) {
   555→      logger.error(`FCM send failed for user ${user.id}:`, err.message);
   556→      if (
   557→        err.code === 'messaging/invalid-registration-token' ||
   558→        err.code === 'messaging/registration-token-not-registered'
   559→      ) {
   560→        await this.supabase.updateFcmToken(user.id, null);
   561→        logger.info(`Cleared invalid FCM token for user ${user.id}`);
   562→      }
   563→      return { channel: 'push', status: 'failed', error: err.message };
   564→    }
   565→  }
   566→
   567→  // ========================================
   568→  // WhatsApp Notifications — PAID (provider-delegated)
   569→  // ========================================
   570→
   571→  async _sendWhatsApp(user, alert) {
   572→    const provider = await this.getActiveProvider();
   573→    if (!provider) {
   574→      return { channel: 'whatsapp', status: 'skipped', reason: 'No notification provider available' };
   575→    }
   576→    if (!provider.isWhatsAppAvailable()) {
   577→      return { channel: 'whatsapp', status: 'skipped', reason: `WhatsApp not configured on ${provider.getProviderName()}` };
   578→    }
   579→    if (!user.phone) {
   580→      return { channel: 'whatsapp', status: 'skipped', reason: 'No phone number' };
   581→    }
   582→    return provider.sendWhatsApp(user, alert);
   583→  }

   586→  // SMS Notifications — PAID (provider-delegated)

   589→  async _sendSMS(user, alert) {
   590→    const provider = await this.getActiveProvider();
   591→    if (!provider) {
   592→      return { channel: 'sms', status: 'skipped', reason: 'No notification provider available' };
   593→    }
   594→    if (!provider.isSmsAvailable()) {
   595→      return { channel: 'sms', status: 'skipped', reason: `SMS not configured on ${provider.getProviderName()}` };
   596→    }
   597→    if (!user.phone) {
   598→      return { channel: 'sms', status: 'skipped', reason: 'No phone number' };
   599→    }
   600→    return provider.sendSMS(user, alert);
   601→  }

   603→  // ========================================
   604→  // Alert Preference Checks
   605→  // ========================================
   606→
   607→  _isAlertEnabled(alertType, user) {
   608→    switch (alertType) {
   609→      case 'geofence_enter':
   610→      case 'geofence_exit':
   611→        return user.alert_geofence_enter !== false && user.alert_geofence_exit !== false;
   612→      case 'battery_low':
   613→        return user.alert_battery_low !== false;
   614→      case 'acc_on':
   615→      case 'acc_off':
   616→        return user.alert_acc_change !== false;
   617→      case 'speed_limit':
   618→        return user.alert_speed_limit !== false;
   619→      // AUDIT 3 — Brouillage GPS/GSM (système legacy global, configuré via NotificationsScreen)
   620→      case 'gps_jamming':
   621→      case 'gps_no_jamming':
   622→        return user.alert_gps_jamming !== false;
   623→      case 'gsm_jamming':
   624→      case 'gsm_no_jamming':
   625→        return user.alert_gsm_jamming !== false;
   626→      // FIX (gps_frozen default OFF) : Contrairement aux autres alertes qui
   627→      // utilisent !== false (default TRUE), gps_frozen utilise === true
   628→      // (default FALSE). L'utilisateur doit explicitement activer cette
   629→      // alerte dans Notifications & Alertes car elle est considérée comme
   630→      // potentiellement gênante (se déclenche après 30 min d'arrêt).
   631→      case 'gps_frozen':
   632→        return user.alert_gps_frozen === true;
   633→      case 'credit_reminder':
   634→      case 'credit_depleted':
   635→        return user.credit_notifications_enabled !== false;
   636→      default:
   637→        return true;
   638→    }
   639→  }

   641→  // ========================================
   642→  // Supabase Realtime Listener
   643→  // ========================================
   644→
   645→  startRealtimeListener() {
   646→    try {
   647→      const channel = this.supabase.client
   648→        .channel('yazz-notification-pusher')
   649→        .on(
   650→          'postgres_changes',
   651→          {
   652→            event: 'INSERT',
   653→            schema: 'public',
   654→            table: 'notifications',
   655→          },
   656→          async (payload) => {
   657→            try {
   658→              const notif = payload.new;
   659→              
   660→              // AUDIT ANTI-SPAM — Le listener ne doit envoyer QUE les notifs
   661→              // qui viennent de pg_cron (crédits) ou des Edge Functions.
   662→              // Toutes les alertes device sont envoyées directement par sendAlert()
   663→              // qui set direct_send=true. Si on les envoie aussi ici → doublon.
   664→              if (notif.data?.direct_send === true) {
   665→                return; // Déjà envoyé — skip silencieux
   666→              }
   667→
   668→              // Seules les notifs de crédit (pg_cron) doivent passer par ici
   669→              const isCreditNotif = notif.type === 'credit_low' || notif.type === 'credit_depleted' || notif.type === 'credit_reminder';
   670→              if (!isCreditNotif) {
   671→                logger.debug(`📬 Realtime: ignoring ${notif.type} (not a credit notification, handled by sendAlert)`);
   672→                return;
   673→              }
   674→              
   675→              logger.info(`📬 Realtime credit notification received: type=${notif.type} user=${notif.user_id}`);
   676→
   677→              // ── Credit gate: block PAID channels only (SMS/WhatsApp) if credits depleted ──
   678→              // Push is always FREE and must never be blocked by credit status.
   679→              // Credit-related notifications always go through regardless.
   680→              const isPaidChannel = notif.channel === 'sms' || notif.channel === 'whatsapp' || notif.type === 'alert_paid';
   681→              let creditsDepleted = false;
   682→              if (isPaidChannel && !isCreditNotif) {
   683→                const creditService = getCreditService();
   684→                const credits = await creditService.getBalance(notif.user_id);
   685→                creditsDepleted = credits && !credits.is_active;
   686→                if (creditsDepleted) {
   687→                  logger.info(`📬 Realtime PAID channel BLOCKED for user ${notif.user_id} — credits depleted, type=${notif.type}`);
   688→                }
   689→              }
   690→
   691→              const user = await this.supabase.getUser(notif.user_id);
   692→
   693→              // Push notification (FCM)
   694→              if (notif.channel === 'push' || notif.type === 'alert_paid') {
   695→                if (user && user.fcm_token) {
   696→                  const result = await this._sendPush(user, {
   697→                    type: notif.type,
   698→                    deviceId: notif.device_id || '',
   699→                    title: notif.title,
   700→                    body: notif.body,
   701→                    data: notif.data || {},
   702→                  });
   703→                  logger.info(`📬 Realtime push result: ${result.status} for user ${notif.user_id}`);
   704→                }
   705→              }
   706→
   707→              // Paid channel notifications from Edge Function
   708→              // Credit gate: skip if credits depleted (push was already sent above)
   709→              if ((notif.channel === 'sms' || notif.channel === 'whatsapp') && !creditsDepleted) {
   710→                const channelToSend = notif.data?.channel_to_send || notif.channel;
   711→                const alert = {
   712→                  type: notif.data?.original_type || notif.type,
   713→                  deviceId: notif.device_id || '',
   714→                  title: notif.title,
   715→                  body: notif.body,
   716→                  data: notif.data || {},
   717→                };
   718→
   719→                if (channelToSend === 'sms' && user) {
   720→                  const result = await this._sendSMS(user, alert);
   721→                  logger.info(`📬 Realtime SMS result: ${result.status} for user ${notif.user_id}`);
   722→                } else if (channelToSend === 'whatsapp' && user) {
   723→                  const result = await this._sendWhatsApp(user, alert);
   724→                  logger.info(`📬 Realtime WhatsApp result: ${result.status} for user ${notif.user_id}`);
   725→                }
   726→              }
   727→
   728→              // Credit-related push notifications (from pg_cron)
   729→              if (notif.type === 'credit_low' || notif.type === 'credit_depleted' || notif.type === 'credit_reminder') {
   730→                if (user && user.fcm_token) {
   731→                  await this._sendPush(user, {
   732→                    type: notif.type,
   733→                    deviceId: notif.device_id || '',
   734→                    title: notif.title,
   735→                    body: notif.body,
   736→                    data: notif.data || {},
   737→                  });
   738→                }
   739→              }
   740→            } catch (err) {
   741→              logger.error('Failed to send realtime notification:', err);
   742→            }
   743→          }
   744→        )
   745→        .subscribe((status) => {
   746→          logger.info(`Supabase Realtime listener status: ${status}`);
   747→        });
   748→
   749→      this._realtimeChannel = channel;
   750→      logger.info('Supabase Realtime notification listener started ✅');
   751→    } catch (err) {
   752→      logger.error('Failed to start Supabase Realtime listener:', err);
   753→    }
   754→  }
   755→
   756→  stopRealtimeListener() {
   757→    if (this._realtimeChannel) {
   758→      this.supabase.client.removeChannel(this._realtimeChannel);
   759→      this._realtimeChannel = null;
   760→      logger.info('Supabase Realtime notification listener stopped');
   761→    }
   762→  }
   763→}

   768→  if (!instance) {
   769→    instance = new NotificationService();
   770→  }
   771→  return instance;
   772→}
